CREATE TABLE dbo.LSVProviderUpdateQueue
(
    QueueId         BIGINT IDENTITY(1,1) NOT NULL PRIMARY KEY,

    BatchId         UNIQUEIDENTIFIER NOT NULL,          -- one run of the job
    InputFileName   NVARCHAR(260)    NOT NULL,
    InputRowNumber  INT              NOT NULL,

    ProviderId      NVARCHAR(50)     NOT NULL,          -- LSVP-ID

    -- Store everything we need without making 500 columns:
    InputRowJson    NVARCHAR(MAX)    NULL,              -- full input row as JSON
    SnapshotJson    NVARCHAR(MAX)    NULL,              -- SQL snapshot as JSON (optional)
    ChangesJson     NVARCHAR(MAX)    NOT NULL,          -- only changed fields list (before/after)

    Status          VARCHAR(20)      NOT NULL DEFAULT ('Pending'), -- Pending/Success/Failed
    AttemptCount    INT              NOT NULL DEFAULT (0),
    LastError       NVARCHAR(4000)   NULL,

    CreatedOn       DATETIME2(0)     NOT NULL DEFAULT (SYSUTCDATETIME()),
    ProcessedOn     DATETIME2(0)     NULL
);

CREATE INDEX IX_LSVProviderUpdateQueue_Status ON dbo.LSVProviderUpdateQueue(Status, CreatedOn);
CREATE INDEX IX_LSVProviderUpdateQueue_ProviderId ON dbo.LSVProviderUpdateQueue(ProviderId);
CREATE INDEX IX_LSVProviderUpdateQueue_BatchId ON dbo.LSVProviderUpdateQueue(BatchId);
