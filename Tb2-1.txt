CREATE TABLE rec10.RawLine (
    RawLineId        BIGINT IDENTITY(1,1) PRIMARY KEY,
    IngestRunId      BIGINT NOT NULL,
    SourceFile       NVARCHAR(260) NOT NULL,
    FileType         NVARCHAR(30) NOT NULL,      -- REC10_OUT / REC10_CHG / CHG_REPORT / RECON_REPORT / MET_ERR / MET_RPT / UNKNOWN
    LineNumber       INT NOT NULL,
    LineLen          INT NOT NULL,
    LineText         NVARCHAR(200) NOT NULL,     -- store as NVARCHAR (safe). You can also store CHAR(80) if you want.
    LineHash         VARBINARY(32) NOT NULL,     -- SHA2_256 for dedupe/audit
    ReceivedAtUtc    DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT FK_RawLine_IngestRun FOREIGN KEY (IngestRunId) REFERENCES rec10.IngestRun(IngestRunId)
);
GO

CREATE INDEX IX_RawLine_RunFileType ON rec10.RawLine (IngestRunId, FileType, SourceFile);
GO
