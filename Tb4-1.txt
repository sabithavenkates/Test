CREATE TABLE rec10.MetErrorRecord (
    MetErrorRecordId BIGINT IDENTITY(1,1) PRIMARY KEY,
    IngestRunId      BIGINT NOT NULL,
    SourceFile       NVARCHAR(260) NOT NULL,
    LineNumber       INT NOT NULL,

    NPI              CHAR(10) NULL,
    MetValue         NVARCHAR(20) NULL,   -- safe; source was X(5)
    RawLine          NVARCHAR(200) NOT NULL,

    ParsedAtUtc      DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME(),

    CONSTRAINT FK_MetErrorRecord_IngestRun FOREIGN KEY (IngestRunId) REFERENCES rec10.IngestRun(IngestRunId)
);
GO
